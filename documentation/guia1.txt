PASO 2.4: Permiso IAM para el rol unauthRole
  En AWS Console (region us-east-2) -> IAM -> Roles -> el rol que termina en
  "unauthRole" -> Add permissions -> Create inline policy -> JSON:

  {
    "Version": "2012-10-17",
    "Statement": [
      {
        "Effect": "Allow",
        "Action": "rekognition:StartFaceLivenessSession",
        "Resource": "*"
      }
    ]
  }

  Nombre sugerido: StartFaceLiveness.

  Nota: si la app Android ya funciona con el mismo rol, este permiso ya existe.
  Verificalo antes de crearlo.

  Requisito previo: el rol unauthRole solo existe si hay un Identity Pool.
  Si no hay ninguno en Cognito (us-east-2), crearlo antes:
    AWS Console (region us-east-2) -> Cognito -> Identity pools -> Create identity pool
      - Activar "Enable access to unauthenticated identities" (guest access)
      - Al terminar, Cognito crea el rol unauthRole automaticamente
      - Copiar el Identity pool ID (formato us-east-2:xxxxxxxx-xxxx-...)

  Luego, en android/app/src/main/res/raw/amplifyconfiguration.json, agregar el
  PoolId dentro de CognitoIdentity.Default (el archivo copiado en PASO 2.3 no lo trae):
    "Default": {
      "PoolId": "us-east-2:xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx",
      "Region": "us-east-2"
    }
  Usar el mismo PoolId en la app Android, si ya la usa.

  Por que no hay login: el óvalo usa identidades anonimas (guest access) de Cognito.
  Cognito entrega credenciales temporales sin usuario ni contrasena. El formulario
  de login de la app es independiente y no interviene en esta comparacion.

PASO 3.4: LivenessActivity.kt
  Ruta: android/app/src/main/java/<paquete>/LivenessActivity.kt

  package <paquete>

  import android.Manifest
  import android.app.Activity
  import android.content.Intent
  import android.content.pm.PackageManager
  import android.os.Bundle
  import androidx.activity.ComponentActivity
  import androidx.activity.compose.setContent
  import androidx.activity.result.contract.ActivityResultContracts
  import androidx.compose.material3.MaterialTheme
  import androidx.core.content.ContextCompat
  import com.amplifyframework.ui.liveness.ui.FaceLivenessDetector

  class LivenessActivity : ComponentActivity() {

      private val cameraLauncher = registerForActivityResult(
          ActivityResultContracts.RequestPermission()
      ) { granted ->
          if (granted) mostrarDetector() else cerrarConError("Permiso de camara denegado")
      }

      override fun onCreate(savedInstanceState: Bundle?) {
          super.onCreate(savedInstanceState)
          if (ContextCompat.checkSelfPermission(this, Manifest.permission.CAMERA)
              == PackageManager.PERMISSION_GRANTED) {
              mostrarDetector()
          } else {
              cameraLauncher.launch(Manifest.permission.CAMERA)
          }
      }

      private fun mostrarDetector() {
          val sessionId = intent.getStringExtra("sessionId")
              ?: return cerrarConError("Sin sessionId")
          val region = intent.getStringExtra("region") ?: "us-east-2"

          setContent {
              MaterialTheme {
                  FaceLivenessDetector(
                      sessionId = sessionId,
                      region = region,
                      onComplete = {
                          setResult(Activity.RESULT_OK, Intent().putExtra("completed", true))
                          finish()
                      },
                      onError = { error ->
                          cerrarConError(error.message ?: "Fallo prueba de vida")
                      }
                  )
              }
          }
      }

      private fun cerrarConError(mensaje: String) {
          setResult(Activity.RESULT_CANCELED, Intent().putExtra("error", mensaje))
          finish()
      }
  }

  Nota: la app Android usa FaceLivenessScreen.kt con el mismo componente.
  Si la firma de FaceLivenessDetector cambio en la version 1.2.6, copia
  exactamente la llamada de FaceLivenessScreen.kt.

PASO 3.5: FaceLivenessPlugin.kt
  Ruta: android/app/src/main/java/<paquete>/FaceLivenessPlugin.kt

  package <paquete>

  import android.app.Activity
  import android.content.Intent
  import android.util.Log
  import androidx.activity.result.ActivityResult
  import com.amplifyframework.auth.cognito.AWSCognitoAuthPlugin
  import com.amplifyframework.core.Amplify
  import com.getcapacitor.JSObject
  import com.getcapacitor.Plugin
  import com.getcapacitor.PluginCall
  import com.getcapacitor.PluginMethod
  import com.getcapacitor.annotation.ActivityCallback
  import com.getcapacitor.annotation.CapacitorPlugin

  @CapacitorPlugin(name = "FaceLiveness")
  class FaceLivenessPlugin : Plugin() {

      companion object {
          private var amplifyListo = false
      }

      override fun load() {
          super.load()
          if (!amplifyListo) {
              try {
                  Amplify.addPlugin(AWSCognitoAuthPlugin())
                  Amplify.configure(context.applicationContext)
                  amplifyListo = true
              } catch (e: Exception) {
                  Log.e("FaceLiveness", "Error configurando Amplify", e)
              }
          }
      }

      @PluginMethod
      fun startLiveness(call: PluginCall) {
          val sessionId = call.getString("sessionId") ?: return call.reject("Falta sessionId")
          val region = call.getString("region", "us-east-2")

          val intent = Intent(context, LivenessActivity::class.java).apply {
              putExtra("sessionId", sessionId)
              putExtra("region", region)
          }
          startActivityForResult(call, intent, "onResultadoLiveness")
      }

      @ActivityCallback
      private fun onResultadoLiveness(call: PluginCall?, result: ActivityResult) {
          if (call == null) return
          if (result.resultCode == Activity.RESULT_OK) {
              call.resolve(JSObject().put("completed", true))
          } else {
              val errorMsg = result.data?.getStringExtra("error") ?: "Cancelado"
              call.reject(errorMsg)
          }
      }
  }

PASO 4.3: Servicio de verificacion
  Crear src/app/services/verificacion-facial.service.ts
  Reemplaza los PEGA_AQUI_...: las URLs de las Lambdas estan en
  Unidad1/app/src/main/java/com/example/tarea_082426/data/remote/ApiLiveness.kt.
  Equivalencia: getSessionId = Lambda 1 (GET), getSessionResults = Lambda 2 (POST),
  compareFaces = Lambda 3 (POST).
  Nota de este proyecto: las URLs ya estan puestas en
  src/app/services/verificacion-facial.service.ts (backend: https://202407391.aplicacionesweb2026.com).
  La URL del backend debe ser accesible desde el celular (no localhost).
  Si es http:// y no https://, agrega android:usesCleartextTraffic="true"
  en <application> de AndroidManifest.xml.

  import { Injectable } from '@angular/core';
  import { HttpClient } from '@angular/common/http';
  import { firstValueFrom } from 'rxjs';
  import { FaceLiveness } from './face-liveness';

  @Injectable({ providedIn: 'root' })
  export class VerificacionFacialService {
    private readonly URL_BACKEND = 'PEGA_AQUI_LA_URL_DEL_BACKEND';
    private readonly URL_GET_SESSION = 'PEGA_AQUI_URL_LAMBDA_1';
    private readonly URL_GET_RESULTS = 'PEGA_AQUI_URL_LAMBDA_2';
    private readonly URL_COMPARE = 'PEGA_AQUI_URL_LAMBDA_3';
    private readonly ID_USUARIO_REFERENCIA = 1;

    constructor(private http: HttpClient) {}

    async verificar(): Promise<{ acceso: boolean; mensaje: string }> {
      // 1. Session ID (Lambda 1)
      const sesion = await firstValueFrom(
        this.http.get<any>(this.URL_GET_SESSION)
      );
      const sessionId =
        typeof sesion === 'string' ? sesion : sesion?.sessionId ?? null;
      if (!sessionId) {
        return { acceso: false, mensaje: 'No se obtuvo sessionId' };
      }

      // 2. Captura con el ovalo de Amplify (sin edicion)
      try {
        await FaceLiveness.startLiveness({ sessionId, region: 'us-east-2' });
      } catch (err: any) {
        return { acceso: false, mensaje: err?.message ?? 'Verificacion cancelada' };
      }

      // 3. Foto en vivo (Lambda 2)
      const resultados = await firstValueFrom(
        this.http.post<any>(this.URL_GET_RESULTS, { sessionId })
      );
      const fotoEnVivo: string | null = resultados?.referenceImageBase64 ?? null;
      if (!fotoEnVivo) {
        return { acceso: false, mensaje: 'No se obtuvo foto en vivo' };
      }

      // 4. Foto de perfil desde la base de datos (id_user = 1)
      const perfil = await firstValueFrom(
        this.http.get<any>(
          `${this.URL_BACKEND}/api/users/${this.ID_USUARIO_REFERENCIA}/profile`
        )
      );
      const fotoPerfil: string | null = perfil?.body?.fotoBase64 ?? null;
      if (!fotoPerfil) {
        return { acceso: false, mensaje: 'El usuario no tiene foto de perfil' };
      }

      // 5. Comparacion (Lambda 3)
      const comparacion = await firstValueFrom(
        this.http.post<any>(this.URL_COMPARE, {
          sourceImageBase64: fotoEnVivo,
          targetImageBase64: fotoPerfil,
        })
      );

      return comparacion?.isMatch
        ? { acceso: true, mensaje: `Coincidencia: ${Number(comparacion.similarity).toFixed(1)}%` }
        : { acceso: false, mensaje: 'El rostro no coincide con la foto registrada' };
    }
  }

  Nota: si fotoBase64 viene con el prefijo "data:image/...;base64,", quitalo
  antes de enviarlo. Verifica el formato real con el GET del backend.
